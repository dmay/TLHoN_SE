cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TLHoN_ItemCatalog LANGUAGES CXX)
    include(CTest)
endif()

find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
add_library(tlhon_item_catalog STATIC itemcatalog.cpp itemcatalog.h)
target_compile_features(tlhon_item_catalog PUBLIC cxx_std_17)
target_include_directories(tlhon_item_catalog PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(tlhon_item_catalog PUBLIC Qt${QT_VERSION_MAJOR}::Core)
add_executable(tlhon-catalog catalog_main.cpp)
target_link_libraries(tlhon-catalog PRIVATE tlhon_item_catalog)
if(BUILD_TESTING)
    add_executable(item_catalog_tests ../tests/item_catalog_tests.cpp)
    target_link_libraries(item_catalog_tests PRIVATE tlhon_item_catalog)
    add_test(NAME item_catalog_tests COMMAND item_catalog_tests
        ${CMAKE_CURRENT_SOURCE_DIR}/.. $<TARGET_FILE:tlhon-catalog>)
endif()
